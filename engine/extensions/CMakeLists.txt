function(extension_shader source name stage entry)
  set(output "${ENGINE_SHADER_OUTPUT_DIR}/${name}.${stage}.spv")
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${ENGINE_SHADER_OUTPUT_DIR}/extensions" "${ENGINE_SHADER_OUTPUT_DIR}/hdr"
    COMMAND "${ENGINE_SLANGC}" "${source}" -target spirv -profile spirv_1_5 -entry ${entry} -stage ${stage} -matrix-layout-column-major -emit-spirv-directly ${ARGN} -o "${output}"
    DEPENDS ${lighting_dependencies} "${PROJECT_SOURCE_DIR}/engine/lighting/shaders/brdf.slang" "${PROJECT_SOURCE_DIR}/engine/lighting/shaders/lighting.slang" "${PROJECT_SOURCE_DIR}/engine/lighting/shaders/DirectionalLight.slang" "${source}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/fullscreen.slang" VERBATIM)
  set(extension_shader_outputs ${extension_shader_outputs} "${output}" PARENT_SCOPE)
endfunction()
