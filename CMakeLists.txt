cmake_minimum_required(VERSION 3.25)
project(Engine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENGINE_ENABLE_PROFILING "Enable Tracy instrumentation" ON)
option(ENGINE_BUILD_EDITOR "Build the live docking editor" ON)
include(CTest)
include(cmake/Dependencies.cmake)
function(engine_target target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
add_subdirectory(engine/core)
add_subdirectory(engine/platform)
add_subdirectory(engine/rhi)
add_subdirectory(engine/renderer)
add_subdirectory(engine/scene)
add_subdirectory(engine/assets)
add_subdirectory(engine/shaders)
add_subdirectory(engine/extensions)
add_subdirectory(engine/lighting)
option(ENGINE_FEATURE_PROCEDURAL_SKY "Include modular procedural sky" ON)
if(ENGINE_FEATURE_PROCEDURAL_SKY)
  add_subdirectory(engine/sky)
endif()
add_subdirectory(apps/runtime)
if(ENGINE_BUILD_EDITOR)
  add_subdirectory(editor)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
